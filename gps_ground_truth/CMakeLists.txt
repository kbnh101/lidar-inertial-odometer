cmake_minimum_required(VERSION 3.16)
project(gps_ground_truth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
add_executable(gps_ground_truth_node src/gps_ground_truth_node.cpp)
target_include_directories(gps_ground_truth_node PRIVATE include)
target_link_libraries(gps_ground_truth_node Eigen3::Eigen)
ament_target_dependencies(gps_ground_truth_node rclcpp sensor_msgs std_msgs nav_msgs)
target_compile_options(gps_ground_truth_node PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_gps_trajectory test/test_trajectory.cpp)
  target_include_directories(test_gps_trajectory PRIVATE include)
  target_link_libraries(test_gps_trajectory Eigen3::Eigen)
endif()
install(TARGETS gps_ground_truth_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
